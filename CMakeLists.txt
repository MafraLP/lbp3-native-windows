    message(FATAL_ERROR "Use clang-cl targeting the Windows MSVC ABI.")

# Avoid a loader <-> HLE cycle: inject the module resolver at runtime.

# Optional native Vulkan/Liverpool backend adapted from the ISS port. Keep it
# opt-in because the source tree and its GPL-2.0-or-later dependencies are
# supplied separately from this repository.
option(LBP3_ENABLE_ISS_GPU "Build and link the ISS native Vulkan GPU backend" OFF)
set(LBP3_ISS_GPU_SOURCE_DIR "${PROJECT_SOURCE_DIR}/out/iss-audit/source/port/gpu" CACHE PATH
    "Path to the audited ISS GPU backend source tree")
if(LBP3_ENABLE_ISS_GPU)
    # The Windows SDL3 port must be built with its `vulkan` feature enabled
    # (vcpkg: sdl3[vulkan]); otherwise SDL_Vulkan_LoadLibrary fails at runtime.
    enable_language(C)
    if(NOT EXISTS "${LBP3_ISS_GPU_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "LBP3_ENABLE_ISS_GPU requires LBP3_ISS_GPU_SOURCE_DIR with CMakeLists.txt")
    endif()
    set(TLG_LTO OFF CACHE BOOL "Disable IPO for the ISS backend in host debug builds" FORCE)
    set(ISS_BUILD_HOST_TARGETS OFF CACHE BOOL "Build only the reusable ISS GPU backend" FORCE)
    add_subdirectory("${LBP3_ISS_GPU_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/iss-gpu")
    target_include_directories(gpu_bridge PRIVATE "${LBP3_ISS_GPU_SOURCE_DIR}"
        "${LBP3_ISS_GPU_SOURCE_DIR}/shadps4")
    target_sources(gpu_bridge PRIVATE src/gpu/iss_jump.cpp
        "${LBP3_ISS_GPU_SOURCE_DIR}/shadps4/common/ntapi.cpp")
    target_link_libraries(gpu_bridge PRIVATE tlggpu)
    target_link_libraries(gpu_bridge PRIVATE onecore ntdll)
    target_compile_definitions(gpu_bridge PRIVATE LBP3_HAS_ISS_GPU=1)
endif()
